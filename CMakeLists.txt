cmake_minimum_required(VERSION 3.16)
project(SensorBridgeX VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SENSORBRIDGEX_WARNINGS "Enable compiler warnings" ON)
add_library(sensorbridgex_core src/config.cpp src/csv.cpp src/simulator.cpp src/validator.cpp)
target_include_directories(sensorbridgex_core PUBLIC include)
add_executable(sensorbridgex src/main.cpp)
target_link_libraries(sensorbridgex PRIVATE sensorbridgex_core)
if(SENSORBRIDGEX_WARNINGS)
  target_compile_options(sensorbridgex_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  target_compile_options(sensorbridgex PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_validator tests/test_validator.cpp)
  target_link_libraries(test_validator PRIVATE sensorbridgex_core)
  add_test(NAME validator_unit_tests COMMAND test_validator)
  add_test(NAME cli_smoke_test COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.sh $<TARGET_FILE:sensorbridgex> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
install(TARGETS sensorbridgex RUNTIME DESTINATION bin)
